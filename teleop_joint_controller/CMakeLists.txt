cmake_minimum_required(VERSION 3.16)
project(teleop_joint_controller LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(urdf REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(controller_interface REQUIRED)
find_package(control_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(realtime_tools REQUIRED)
add_library(${PROJECT_NAME} SHARED src/teleop_joint_controller.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include/${PROJECT_NAME}>)
ament_target_dependencies(${PROJECT_NAME} urdf rcl_interfaces controller_interface control_msgs diagnostic_msgs moveit_msgs trajectory_msgs hardware_interface pluginlib rclcpp rclcpp_lifecycle realtime_tools)
pluginlib_export_plugin_description_file(controller_interface teleop_joint_controller.xml)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_reference_filter test/test_reference_filter.cpp)
  target_link_libraries(test_reference_filter ${PROJECT_NAME})
  ament_add_gtest(test_teleop_joint_controller test/test_teleop_joint_controller.cpp)
  target_link_libraries(test_teleop_joint_controller ${PROJECT_NAME})
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(controller_interface control_msgs diagnostic_msgs moveit_msgs trajectory_msgs hardware_interface realtime_tools rclcpp rclcpp_lifecycle)
ament_package()
